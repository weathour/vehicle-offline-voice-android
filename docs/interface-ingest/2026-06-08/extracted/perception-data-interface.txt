车道线数据

数据内容                 Key                 值类型            备注

车道线数据                Sensor_Lanelist     LaneList       ﻿


                                                             复制代码

syntax = "proto3";

message Pt {
    double x = 1;          // X 坐标
    double y = 2;          // Y 坐标
}

message Line {
    int32 lane_type = 1;                // 车道线类型
    int32 pts_vehicle_num = 2;
    repeated Pt pts_vehicle = 3;        // 车辆坐标系车道线点集
    int32 pts_abs_num = 4;
    repeated Pt pts_abs = 5;            // 世界坐标系车道线点集
    double confidence = 6;              // 车道线置信度
}

message LaneList {
    double timestamp = 1;               // 时间戳
    int32 lane_num = 2;
    repeated Line laneline = 3;
}
